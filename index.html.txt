            const fetchPatients = async () => {
                if(isSyncingRef.current || API_URL === 'PASTE_YOUR_WEB_APP_URL_HERE') return; 

                try {
                    const response = await fetch(API_URL);
                    const textData = await response.text(); 
                    
                    try {
                        const data = JSON.parse(textData); 
                        if (Array.isArray(data)) {
                            setPatients(data);
                        } else {
                            setPatients(data.patients || []);
                            if(data.passwords) setPasswords(data.passwords);
                        }
                    } catch (jsonError) {
                        console.error("Data Parse Error", textData);
                    }
                } catch (error) {
                    console.error("Network Fetch error:", error);
                } finally {
                    setIsInitialLoad(false);
                }
            };

            const syncToDB = (action, payload) => {
                if(API_URL === 'PASTE_YOUR_WEB_APP_URL_HERE') return;
                setIsSyncing(true);
                isSyncingRef.current = true;
                
                fetch(API_URL, {
                    method: 'POST',
                    headers: { 'Content-Type': 'text/plain;charset=utf-8' },
                    body: JSON.stringify({ action, ...payload })
                }).finally(() => {
                    setTimeout(() => {
                        setIsSyncing(false);
                        isSyncingRef.current = false;
                    }, 2500); 
                });
            };

            React.useEffect(() => {
                fetchPatients(); 
                const clockInterval = setInterval(() => setCurrentTime(Date.now()), 60000);
                return () => clearInterval(clockInterval);
            }, []);

            React.useEffect(() => {
                if (userRole) {
                    const pollInterval = setInterval(fetchPatients, 5000);
                    return () => clearInterval(pollInterval);
                }
            }, [userRole]);
            
            React.useEffect(() => {
                if (isInitialLoad || !userRole) return;
                
                let shouldChime = false;
                const myAlertRole = (userRole === 'LR' || userRole === 'Admin') ? 'LR' : userRole; 

                patients.forEach(p => {
                    const prevP = prevPatientsRef.current.find(old => old.id === p.id);
                    const hasNewAlert = p.lastVE?.alertRole === myAlertRole && (!prevP || prevP.lastVE?.alertRole !== myAlertRole);
                    const isNewlyOverdue = (myAlertRole === 'LR' && p.lastVE?.nextReviewTime && currentTime > new Date(p.lastVE.nextReviewTime).getTime() && !p.lastVE?.reviewAcked) && 
                                           (!prevP || !(prevP.lastVE?.nextReviewTime && currentTime > new Date(prevP.lastVE.nextReviewTime).getTime() && !prevP.lastVE?.reviewAcked));
                    
                    if (hasNewAlert || isNewlyOverdue) shouldChime = true;
                });

                if (shouldChime) playAlertSound('new_booking');
                prevPatientsRef.current = patients;
            }, [patients, currentTime, isInitialLoad, userRole]);

            const calculatePriority = (patient) => {
                let score = 0;
                
                // IOL Priority (+10 for IOL, 0 for spontaneous)
                if (patient.admissionType && patient.admissionType.includes('IOL')) score += 10;
                
                // CTG
                if (patient.ctg === 'Pathological') score += 100;
                if (patient.ctg === 'Suspicious') score += 50;
                
                // Liquor
                if (patient.liquor === 'TMSL' || patient.liquor === 'Blood stained') score += 40;
                else if (patient.liquor === 'MMSL') score += 30;
                else if (patient.liquor === 'LMSL') score += 15;

                // GBS & ROM (+30 stacked if both)
                const hasRom = ['SROM', 'ARM'].includes(patient.membrane) || patient.membrane === 'PROM_MI';
                if (hasRom) {
                    score += 15;
                    if (patient.gbs) score += 15; 
                }
                
                // Dilation (+40 for imminent delivery)
                if (patient.lastVE) {
                    if (patient.lastVE.dilation >= 9) score += 40;
                    else if (patient.lastVE.dilation >= 7) score += 20;
                }

                // Previous Scar
                if (patient.issues && patient.issues.some(i => typeof i === 'string' && i.includes('1 Prev Scar')) && patient.lastVE && patient.lastVE.dilation > 4) {
                    score += 15;
                }
                
                return score + (patient.manualOverride || 0);
            };

            const formatWaitTime = (bookTime) => {
                const diffMins = Math.floor((currentTime - new Date(bookTime).getTime()) / 60000);
                const days = Math.floor(diffMins / 1440);
                const hrs = Math.floor((diffMins % 1440) / 60);
                const mins = diffMins % 60;
                
                if (days > 0) return `${days}d ${hrs}h ${mins}m`;
                if (hrs > 0) return `${hrs}h ${mins}m`;
                return `${mins}m`;
            };

            const showToast = (message, type = 'info') => {
                setToast({ message, type });
                setTimeout(() => setToast(null), 4000);
            };

            const handleUpdatePatient = (updatedPatient) => {
                setPatients(prev => prev.map(p => p.id === updatedPatient.id ? updatedPatient : p));
                syncToDB('UPDATE', { patient: updatedPatient });
            };

            const handleUpdatePasswords = (newPasswords) => {
                setPasswords(newPasswords);
                syncToDB('UPDATE_PASSWORDS', { passwords: newPasswords });
                showToast('Passwords updated successfully', 'success');
                setIsSettingsOpen(false);
            };

            const handleConfirmBedAssignment = (bed) => {
                const updatedPatient = { ...assignBedPatient, status: 'CALLED', calledToBed: bed, callTime: new Date().toISOString() };
                updatedPatient.lastVE = { ...updatedPatient.lastVE, alertRole: updatedPatient.location, alertMsg: `CALLED TO ${bed}` };
                
                handleUpdatePatient(updatedPatient);
                playAlertSound('called');
                showToast(`${assignBedPatient.name} called to ${bed}. Awaiting Ward transfer.`, 'success');
                setAssignBedPatient(null);
            };

            const handleConfirmMoveBed = (bed) => {
                const updatedPatient = { ...moveBedPatient, assignedBed: bed };
                handleUpdatePatient(updatedPatient);
                showToast(`Patient successfully moved to ${bed}`, 'success');
                setMoveBedPatient(null);
            };

            const handleWardSend = (id, staffName) => {
                const patient = patients.find(p => p.id === id);
                const updatedPatient = { ...patient, status: 'EN_ROUTE', wardSendTime: new Date().toISOString(), wardNotedBy: staffName };
                updatedPatient.lastVE = { ...updatedPatient.lastVE, alertRole: 'LR', alertMsg: `EN ROUTE (Sent by ${staffName})` };
                
                handleUpdatePatient(updatedPatient);
                playAlertSound('update');
                showToast('Transfer confirmed. Patient en route to LR.', 'success');
            };

            const handlePatientArrived = (id) => {
                const patient = patients.find(p => p.id === id);
                const updatedPatient = { ...patient, status: 'IN_LR', assignedBed: patient.calledToBed, admissionTime: new Date().toISOString() };
                handleUpdatePatient(updatedPatient);
                playAlertSound('update');
                showToast('Patient arrived. LR timer started.', 'success');
            };

            const handleSendToOT = (id) => {
                const patient = patients.find(p => p.id === id);
                const updatedPatient = { ...patient, status: 'IN_LR', assignedBed: 'OT' };
                handleUpdatePatient(updatedPatient);
                playAlertSound('critical');
                showToast('URGENT: Patient transferred to OT', 'warning');
            };

            const handleConfirmDelivery = (time) => {
                const today = new Date();
                const [hours, minutes] = time.split(':');
                today.setHours(parseInt(hours, 10), parseInt(minutes, 10), 0, 0);
                
                const pad = (n) => String(n).padStart(2, '0');
                const deliveryISO = `${today.getFullYear()}-${pad(today.getMonth() + 1)}-${pad(today.getDate())}T${pad(today.getHours())}:${pad(today.getMinutes())}:00`;
                
                const updatedPatient = { 
                    ...deliverModalPatient, 
                    status: 'DELIVERED', 
                    deliveryTime: deliveryISO,
                    lastVE: { ...deliverModalPatient.lastVE, alertRole: null, alertMsg: null, reviewAcked: true } 
                };
                handleUpdatePatient(updatedPatient);
                showToast('Delivery recorded. Bed can be transferred when patient leaves.', 'success');
                setDeliverModalPatient(null);
            };

            const handleClearBed = (id) => {
                const patient = patients.find(p => p.id === id);
                const updatedPatient = { 
                    ...patient, 
                    status: 'DISCHARGED',
                    lastVE: { ...patient.lastVE, alertRole: null, alertMsg: null, reviewAcked: true }
                };
                setPatients(prev => prev.map(p => p.id === id ? updatedPatient : p));
                syncToDB('UPDATE', { patient: updatedPatient });
                showToast('Patient transferred to Ward. Record saved to archive.', 'info');
            };

            const handleCancelBooking = (id) => {
                const patient = patients.find(p => p.id === id);
                const updatedPatient = { 
                    ...patient, 
                    status: 'CANCELLED',
                    lastVE: { ...patient.lastVE, alertRole: null, alertMsg: null, reviewAcked: true }
                };
                setPatients(prev => prev.map(p => p.id === id ? updatedPatient : p));
                syncToDB('UPDATE', { patient: updatedPatient });
                showToast('Booking cancelled. Record saved to archive.', 'info');
            };

            const handleManualBumpUp = (id) => {
                const patient = patients.find(p => p.id === id);
                const updatedPatient = { ...patient, manualOverride: (patient.manualOverride || 0) + 10 };
                handleUpdatePatient(updatedPatient);
                playAlertSound('update');
                showToast('Priority nudged UP', 'success');
            };

            const handleManualBumpDown = (id) => {
                const patient = patients.find(p => p.id === id);
                const updatedPatient = { ...patient, manualOverride: (patient.manualOverride || 0) - 10 };
                handleUpdatePatient(updatedPatient);
                playAlertSound('update');
                showToast('Priority nudged DOWN', 'info');
            };

            const handleAcknowledgeAlert = (id, staffName, alertType) => {
                const patient = patients.find(p => p.id === id);
                const updatedPatient = { ...patient };
                
                if (alertType === 'NORMAL') {
                    if (updatedPatient.lastVE?.alertMsg === 'NEW BOOKING') {
                        updatedPatient.notedBy = staffName;
                        updatedPatient.notedTime = new Date().toISOString();
                    }
                    updatedPatient.lastVE = { 
                        ...updatedPatient.lastVE, 
                        alertRole: null, 
                        alertMsg: null,
                        lastGeneralAckBy: staffName,
                        lastGeneralAckTime: new Date().toISOString()
                    };
                } 
                else if (alertType === 'REVIEW') {
                    updatedPatient.lastVE = { 
                        ...updatedPatient.lastVE, 
                        reviewAcked: true,
                        reviewAckBy: staffName,
                        reviewAckTime: new Date().toISOString()
                    };
                }
                
                handleUpdatePatient(updatedPatient);
                playAlertSound('update');
                showToast(`Alert acknowledged by ${staffName}`, 'success');
            };

            const handleNewBooking = (newPatient) => {
                newPatient.lastVE = { ...newPatient.lastVE, alertRole: 'LR', alertMsg: 'NEW BOOKING' };
                setPatients(prev => [...prev, newPatient]);
                syncToDB('ADD', { patient: newPatient });
                setIsBookingModalOpen(false);
                playAlertSound('new_booking');
                showToast('New patient booked. Awaiting LR acknowledgement.', 'success');
            };

            const handleStatusUpdate = (updatedData) => {
                const patientToUpdate = patients.find(p => p.id === updatedData.id);
                const newPatient = { ...patientToUpdate, ...updatedData, manualOverride: 0 }; 
                
                if (userRole !== 'LR' && userRole !== 'Admin') {
                    newPatient.lastVE = { ...newPatient.lastVE, alertRole: 'LR', alertMsg: 'CLINICAL STATUS UPDATED' };
                } else {
                    newPatient.lastVE = { ...newPatient.lastVE, reviewAcked: false, alertRole: null, alertMsg: null };
                }
                
                handleUpdatePatient(newPatient);
                setUpdateModalPatient(null);
                playAlertSound('update');
                showToast(`Clinical status updated`, 'success');
            };

            const waitingPatients = patients.filter(p => p.status === 'WAITING' || p.status === 'CALLED').sort((a, b) => {
                if (a.status === 'CALLED' && b.status !== 'CALLED') return -1;
                if (b.status === 'CALLED' && a.status !== 'CALLED') return 1;
                if (a.status === 'CALLED' && b.status === 'CALLED') return new Date(a.callTime).getTime() - new Date(b.callTime).getTime();

                const scoreDiff = calculatePriority(b) - calculatePriority(a);
                if (scoreDiff !== 0) return scoreDiff;
                return new Date(a.bookingTime).getTime() - new Date(b.bookingTime).getTime(); 
            });

            const defaultPasswords = { 'LR': 'lr123', 'PAC': 'pac123', '7A': '7a123', '7B': '7b123', 'Admin': 'admin123' };
            const activePasswords = passwords || defaultPasswords;

            if (isInitialLoad && !passwords) {
                 return (
                    <div className="loading-overlay">
                        <i className="ph-fill ph-hourglass-high animate-spin text-6xl text-blue-600 mb-4"></i>
                        <h2 className="text-xl font-bold text-slate-800">Retrieving Live Board...</h2>
                        <p className="text-sm text-slate-500 mt-2">Connecting to secure database</p>
                    </div>
                );
            }

            if (!userRole) {
                return <LoginScreen passwords={activePasswords} onLogin={(role) => {
                    setUserRole(role);
                    if (audioCtx.state === 'suspended') audioCtx.resume();
                }} />;
            }

            const isLR = userRole === 'LR' || userRole === 'Admin';

            return (
                <div className="min-h-screen flex flex-col h-screen overflow-hidden">
                    <header className="bg-slate-900 text-white p-3 shadow-md flex justify-between items-center z-10 shrink-0">
                        <div className="flex items-center gap-3">
                            <i className="ph-fill ph-heartbeat text-3xl text-rose-500"></i>
                            <div>
                                <h1 className="text-xl font-bold leading-tight">LR Smart Board</h1>
                                <p className="text-xs text-slate-400">O&G Hospital Tengku Permaisuri Norashikin</p>
                            </div>
                        </div>
                        <div className="flex flex-col items-end gap-1.5">
                            <div className="flex items-center gap-3">
                                <span className="text-sm font-semibold text-slate-300">
                                    Logged in as: <span className="text-white">{isLR ? (userRole === 'Admin' ? 'System Admin' : 'Labour Room Staff') : `Ward ${userRole} Staff`}</span>
                                </span>
                                {userRole === 'Admin' && (
                                    <button onClick={() => setIsSettingsOpen(true)} className="bg-slate-700 hover:bg-slate-600 text-white px-3 py-1.5 rounded font-medium flex items-center gap-1 shadow transition text-xs">
                                        <i className="ph-bold ph-gear"></i> Passwords
                                    </button>
                                )}
                                {!isLR && (
                                    <button onClick={() => setIsBookingModalOpen(true)} className="bg-blue-600 hover:bg-blue-500 text-white px-4 py-1.5 rounded font-medium flex items-center gap-2 shadow-lg transition text-sm">
                                        <i className="ph-bold ph-plus"></i> Book Bed
                                    </button>
                                )}
                                <button onClick={() => {setUserRole(null); setPatients([]);}} className="text-slate-400 hover:text-white px-2 py-1 rounded text-sm transition">Logout</button>
                            </div>
                            <div className="flex gap-2">
                                {isSyncing && (
                                    <div className="text-[11px] font-bold text-amber-400 flex items-center gap-1 bg-amber-900/40 px-2 py-0.5 rounded border border-amber-700">
                                        <i className="ph-bold ph-arrows-clockwise animate-spin"></i> SYNCING...
                                    </div>
                                )}
                                <div className="text-[12px] font-mono text-emerald-400 font-bold flex items-center gap-1.5 bg-slate-800/80 px-2 py-0.5 rounded border border-slate-700">
                                    <i className="ph ph-clock"></i>
                                    {new Date(currentTime).toLocaleString('en-GB', { weekday: 'short', day: 'numeric', month: 'short', year: 'numeric', hour: '2-digit', minute: '2-digit', hour12: true }).toUpperCase()}
                                </div>
                            </div>
                        </div>
                    </header>

                    {toast && (
                        <div className={`fixed top-20 right-4 z-50 p-4 rounded-lg shadow-xl text-white font-medium flex items-center gap-3 animate-bounce ${toast.type === 'warning' ? 'bg-rose-500' : 'bg-emerald-500'}`}>
                            <i className={`ph-fill ${toast.type === 'warning' ? 'ph-warning' : 'ph-check-circle'} text-2xl`}></i> {toast.message}
                        </div>
                    )}

                    <div className="flex-1 flex overflow-hidden">
                        <div className="w-1/3 bg-slate-200/40 border-r border-slate-300 flex flex-col">
                            <div className="p-3 bg-white/50 border-b border-slate-300 flex justify-between items-center shrink-0">
                                <h2 className="font-bold text-slate-800 text-base flex items-center gap-2">
                                    <i className="ph-fill ph-list-numbers text-amber-500"></i> Waiting Queue
                                </h2>
                                <span className="bg-slate-700 text-white text-xs px-2 py-1 rounded-full font-bold">{waitingPatients.length} Waiting</span>
                            </div>
                            
                            <div className="flex-1 overflow-y-auto p-3 custom-scrollbar flex flex-col gap-3">
                                {waitingPatients.map((patient, index) => (
                                    <PatientCard 
                                        key={patient.id} 
                                        patient={patient} 
                                        userRole={userRole}
                                        priorityScore={calculatePriority(patient)}
                                        waitTime={formatWaitTime(patient.bookingTime)}
                                        queueNumber={index + 1}
                                        onCall={() => setAssignBedPatient(patient)}
                                        onUpdate={() => setUpdateModalPatient(patient)}
                                        onBumpUp={() => handleManualBumpUp(patient.id)}
                                        onBumpDown={() => handleManualBumpDown(patient.id)}
                                        onCancel={() => handleCancelBooking(patient.id)}
                                        onAcknowledge={handleAcknowledgeAlert}
                                        onWardSend={handleWardSend}
                                        currentTime={currentTime}
                                    />
                                ))}
                                {waitingPatients.length === 0 && (
                                    <div className="text-center text-slate-400 mt-10 flex flex-col items-center">
                                        <i className="ph-fill ph-check-circle text-4xl mb-2 text-slate-300"></i>
                                        <p>Queue is empty.</p>
                                    </div>
                                )}
                            </div>
                        </div>

                        <div className="w-2/3 bg-slate-100 flex flex-col">
                            <div className="p-3 bg-white/50 border-b border-slate-200 shrink-0 flex justify-between items-center">
                                <h2 className="font-bold text-indigo-900 text-base flex items-center gap-2">
                                    <i className="ph-fill ph-layout text-indigo-500"></i> Labour Room Board
                                </h2>
                            </div>
                            <div className="flex-1 p-4 overflow-y-auto custom-scrollbar">
                                <div className="grid grid-cols-2 gap-6">
                                    <div className="flex flex-col gap-3">
                                        {LR_BEDS_LEFT.map(bed => {
                                            const occupant = patients.find(p => p.assignedBed === bed && (p.status === 'IN_LR' || p.status === 'DELIVERED'));
                                            const incoming = patients.find(p => p.calledToBed === bed && (p.status === 'CALLED' || p.status === 'EN_ROUTE'));
                                            return <BedSlot key={bed} bedName={bed} patient={occupant} incoming={incoming} isLR={isLR} currentTime={currentTime} onUpdate={() => setUpdateModalPatient(occupant)} onDeliver={() => setDeliverModalPatient(occupant)} onSendToOT={() => handleSendToOT(occupant?.id)} onClear={() => handleClearBed(occupant?.id)} onArrived={handlePatientArrived} onAcknowledge={handleAcknowledgeAlert} onMoveBed={() => setMoveBedPatient(occupant)} />;
                                        })}
                                    </div>
                                    <div className="flex flex-col gap-3">
                                        {LR_BEDS_RIGHT.map(bed => {
                                            const occupant = patients.find(p => p.assignedBed === bed && (p.status === 'IN_LR' || p.status === 'DELIVERED'));
                                            const incoming = patients.find(p => p.calledToBed === bed && (p.status === 'CALLED' || p.status === 'EN_ROUTE'));
                                            return <BedSlot key={bed} bedName={bed} patient={occupant} incoming={incoming} isLR={isLR} currentTime={currentTime} onUpdate={() => setUpdateModalPatient(occupant)} onDeliver={() => setDeliverModalPatient(occupant)} onSendToOT={() => handleSendToOT(occupant?.id)} onClear={() => handleClearBed(occupant?.id)} onArrived={handlePatientArrived} onAcknowledge={handleAcknowledgeAlert} onMoveBed={() => setMoveBedPatient(occupant)} />;
                                        })}
                                    </div>
                                </div>
                            </div>
                        </div>
                    </div>

                    {isSettingsOpen && <SettingsModal passwords={activePasswords} onClose={() => setIsSettingsOpen(false)} onSave={handleUpdatePasswords} />}
                    {isBookingModalOpen && <BookingForm userWard={userRole} onClose={() => setIsBookingModalOpen(false)} onSave={handleNewBooking} />}
                    {updateModalPatient && <UpdateForm patient={updateModalPatient} onClose={() => setUpdateModalPatient(null)} onSave={handleStatusUpdate} />}
                    
                    {assignBedPatient && <AssignBedModal patient={assignBedPatient} activePatients={patients} onClose={() => setAssignBedPatient(null)} onConfirm={handleConfirmBedAssignment} />}
                    {moveBedPatient && <AssignBedModal patient={moveBedPatient} activePatients={patients} onClose={() => setMoveBedPatient(null)} onConfirm={handleConfirmMoveBed} isMoveMode={true} />}
                    
                    {deliverModalPatient && <DeliveryModal patient={deliverModalPatient} onClose={() => setDeliverModalPatient(null)} onConfirm={handleConfirmDelivery} />}
                </div>
            );
        };

        const LoginScreen = ({ onLogin, passwords }) => {
            const handleLogin = (e) => {
                e.preventDefault();
                const role = e.target.role.value;
                const pwd = e.target.password.value;
                if (passwords[role] && pwd === passwords[role]) onLogin(role);
                else alert('Incorrect password');
            };

            return (
                <div className="min-h-screen bg-slate-100 flex items-center justify-center">
                    <div className="bg-white p-8 rounded-xl shadow-2xl w-full max-w-md border border-slate-200">
                        <div className="text-center mb-6">
                            <i className="ph-fill ph-heartbeat text-5xl text-rose-500 mb-2"></i>
                            <h1 className="text-2xl font-bold text-slate-800">LR Smart Board</h1>
                            <p className="text-sm text-slate-500">Hospital Tengku Permaisuri Norashikin</p>
                        </div>
                        <form onSubmit={handleLogin} className="space-y-4">
                            <div>
                                <label className="block text-sm font-bold text-slate-700 mb-1">Select Department</label>
                                <select name="role" className="w-full border border-slate-300 rounded-lg p-2.5 bg-slate-50">
                                    <option value="LR">Labour Room (LR)</option>
                                    <option value="PAC">PAC</option>
                                    <option value="7A">Ward 7A</option>
                                    <option value="7B">Ward 7B</option>
                                    <option value="Admin">System Admin</option>
                                </select>
                            </div>
                            <div>
                                <label className="block text-sm font-bold text-slate-700 mb-1">Password</label>
                                <input required type="password" name="password" className="w-full border border-slate-300 rounded-lg p-2.5 bg-slate-50" />
                            </div>
                            <button type="submit" className="w-full bg-blue-600 hover:bg-blue-700 text-white font-bold py-2.5 rounded-lg transition shadow-md">
                                Access Board
                            </button>
                        </form>
                    </div>
                </div>
            );
        };

        const SettingsModal = ({ passwords, onClose, onSave }) => {
            const [localPwds, setLocalPwds] = React.useState(passwords);

            const handleSubmit = (e) => {
                e.preventDefault();
                onSave(localPwds);
            };

            return (
                <div className="fixed inset-0 bg-slate-900/60 backdrop-blur-sm flex items-center justify-center p-4 z-50">
                    <div className="bg-white rounded-xl shadow-2xl w-full max-w-sm p-6 relative">
                        <button onClick={onClose} className="absolute top-4 right-4 text-slate-400 hover:text-slate-600"><i className="ph ph-x text-xl"></i></button>
                        <h2 className="text-lg font-bold text-slate-800 mb-4 flex items-center gap-2"><i className="ph-fill ph-lock-key text-indigo-500"></i> Manage Passwords</h2>
                        <form onSubmit={handleSubmit} className="space-y-3">
                            {Object.entries(localPwds).map(([role, pwd]) => (
                                <div key={role} className="flex items-center gap-3">
                                    <label className="w-20 text-sm font-bold text-slate-600">{role}</label>
                                    <input type="text" value={pwd} onChange={(e) => setLocalPwds({...localPwds, [role]: e.target.value})} className="flex-1 border border-slate-300 rounded p-2 text-sm bg-slate-50 focus:bg-white" required />
                                </div>
                            ))}
                            <div className="pt-4 border-t mt-4 flex justify-end gap-2">
                                <button type="button" onClick={onClose} className="px-4 py-2 text-sm font-semibold text-slate-500 hover:bg-slate-100 rounded transition">Cancel</button>
                                <button type="submit" className="px-4 py-2 text-sm font-semibold bg-indigo-600 text-white rounded hover:bg-indigo-500 shadow transition">Save Passwords</button>
                            </div>
                        </form>
                    </div>
                </div>
            );
        };

        const PatientCard = ({ patient, userRole, priorityScore, waitTime, queueNumber, onCall, onUpdate, onBumpUp, onBumpDown, onCancel, onAcknowledge, onWardSend, currentTime }) => {
            const [ackName, setAckName] = React.useState('');
            const [wardAckName, setWardAckName] = React.useState('');
            
            const isLR = userRole === 'LR' || userRole === 'Admin';
            const canEdit = isLR || patient.location === userRole;
            
            let urgencyClass = 'urgency-low';
            if (priorityScore > 40) urgencyClass = 'urgency-high';

        const UpdateForm = ({ patient, onClose, onSave }) => {
            const [membraneStatus, setMembraneStatus] = React.useState(patient.membrane);
            const handleSubmit = (e) => {
                e.preventDefault();
                const fd = new FormData(e.target);
                const membrane = fd.get('membrane');
                const veTime = fd.get('veTime');
                const romTime = membrane === 'Intact' ? null : (fd.get('romTime') || null);
                
                const reviewInterval = fd.get('reviewInterval');
                let nextReviewTime = patient.lastVE?.nextReviewTime; 
                
                if (reviewInterval === 'CLEAR') {
                    nextReviewTime = null;
                } else if (reviewInterval) {
                    const veDate = new Date(veTime);
                    veDate.setHours(veDate.getHours() + parseInt(reviewInterval, 10));
                    const pad = (n) => String(n).padStart(2, '0');
                    nextReviewTime = `${veDate.getFullYear()}-${pad(veDate.getMonth() + 1)}-${pad(veDate.getDate())}T${pad(veDate.getHours())}:${pad(veDate.getMinutes())}:00`;
                }

                onSave({
                    id: patient.id,
                    lastVE: { 
                        time: veTime, 
                        dilation: fd.get('veDilation'), 
                        cervix: fd.get('veCervix'),
                        nextReviewTime: nextReviewTime
                    },
                    membrane, liquor: membrane === 'Intact' ? 'N/A' : (fd.get('liquor') || 'N/A'), romTime, ctg: fd.get('ctg'),
                    epidural: fd.get('epidural') === 'on'
                });
            };
            return (
                <div className="fixed inset-0 bg-slate-900/60 backdrop-blur-sm flex items-center justify-center p-4 z-50">
                    <div className="bg-white rounded-xl shadow-2xl w-full max-w-lg p-6 relative">
                        <button onClick={onClose} className="absolute top-4 right-4 text-slate-400 hover:text-slate-600"><i className="ph ph-x text-xl"></i></button>
                        <h2 className="text-lg font-bold text-slate-800 mb-1">Update Clinical Status</h2>
                        <p className="text-sm text-slate-500 mb-4 pb-2 border-b">{patient.name}</p>
                        <form onSubmit={handleSubmit} className="space-y-4">
                            <div className="grid grid-cols-4 gap-4 items-end">
                                <div className="col-span-2"><label className="block text-xs font-bold text-slate-600 mb-1">VE Date/Time</label><CustomDateTimePicker name="veTime" defaultValue={patient.lastVE?.time} /></div>
                                <div><label className="block text-xs font-bold text-slate-600 mb-1">Os (cm)</label><input required name="veDilation" type="number" min="0" max="10" defaultValue={patient.lastVE?.dilation} className="w-full border rounded p-1.5 text-xs bg-slate-50" /></div>
                                <div><label className="block text-xs font-bold text-slate-600 mb-1">Cervix</label><input required name="veCervix" type="text" defaultValue={patient.lastVE?.cervix} className="w-full border rounded p-1.5 text-xs bg-slate-50" /></div>
                            </div>
                            <div>
                                <label className="block text-[11px] font-bold text-blue-800 mb-1">Plan Next Review In:</label>
                                <select name="reviewInterval" className="w-full border border-blue-300 rounded p-1.5 text-xs font-semibold text-blue-900 bg-blue-50">
                                    <option value="">Keep Current Timer</option>
                                    <option value="CLEAR">Remove Timer</option>
                                    <option value="1">1 Hour</option>
                                    <option value="2">2 Hours</option>
                                    <option value="3">3 Hours</option>
                                    <option value="4">4 Hours</option>
                                </select>
                            </div>
                            <div className="grid grid-cols-4 gap-3 items-end">
                                <div>
                                    <label className="block text-xs font-bold text-slate-600 mb-1">Membrane</label>
                                    <select name="membrane" value={membraneStatus} onChange={(e) => setMembraneStatus(e.target.value)} className="w-full border rounded p-1.5 text-[11px] bg-slate-50">
                                        <option value="Intact">Intact (MI)</option><option value="SROM">Spont. (SROM)</option><option value="PROM_MI">PROM/SROM</option><option value="ARM">Artificial (ARM)</option>
                                    </select>
                                </div>
                                <div className="col-span-2">

                                    <CustomDateTimePicker name="romTime" defaultValue={patient.romTime} disabled={membraneStatus === 'Intact'} />
                                </div>
                                <div>
                                    <label className={`block text-xs font-bold mb-1 ${membraneStatus === 'Intact' ? 'text-slate-400' : 'text-slate-600'}`}>Liquor</label>
                                    <select name="liquor" defaultValue={patient.liquor} disabled={membraneStatus === 'Intact'} className="w-full border rounded p-1.5 text-xs bg-slate-50 disabled:opacity-50">
                                        <option value="N/A">N/A</option><option value="Clear">Clear</option><option value="LMSL">LMSL</option><option value="MMSL">MMSL</option><option value="TMSL">TMSL</option><option value="Blood stained">Blood stained</option>
                                    </select>
                                </div>
                            </div>
                            <div>
                                <label className="block text-xs font-bold text-slate-600 mb-1">CTG Status</label>
                                <select name="ctg" defaultValue={patient.ctg} className="w-full border rounded p-1.5 text-xs font-semibold bg-slate-50">
                                    <option value="Normal">Normal</option><option value="Suspicious">Suspicious</option><option value="Pathological">Pathol.</option>
                                </select>
                            </div>

                                <label className="flex items-center gap-2 text-xs font-semibold text-blue-700 bg-blue-50 px-2 py-1.5 rounded border border-blue-100 cursor-pointer hover:bg-blue-100 transition">
                                    <input type="checkbox" name="epidural" defaultChecked={patient.epidural} className="w-4 h-4 text-blue-600 rounded" /> 
                                    Epidural (Active / Requested)
                                </label>
                            </div>
                            <div className="flex justify-end gap-3 pt-4">
                                <button type="button" onClick={onClose} className="px-4 py-2 text-sm font-semibold text-slate-500 hover:text-slate-700">Cancel</button>

                            </div>
                        </form>
                    </div>
                </div>
            );
        };
